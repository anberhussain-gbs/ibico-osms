UserAccounts\n```sql\nCREATE TABLE UserAccounts (\n    UserId INT PRIMARY KEY IDENTITY(1,1),\n    Email NVARCHAR(255) NOT NULL UNIQUE,\n    PasswordHash NVARCHAR(MAX) NOT NULL,\n    FirstName NVARCHAR(100) NOT NULL,\n    LastName NVARCHAR(100),\n    IsActive BIT NOT NULL DEFAULT 1,\n    LastLoginDate DATETIME,\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    UpdatedAt DATETIME NOT NULL DEFAULT GETUTCDATE()\n);\n\nCREATE CLUSTERED INDEX IX_UserAccounts_Email ON UserAccounts(Email);\n```\n\n### 2. Roles\n```sql\nCREATE TABLE Roles (\n    RoleId INT PRIMARY KEY IDENTITY(1,1),\n    RoleName NVARCHAR(100) NOT NULL UNIQUE,\n    Description NVARCHAR(500),\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE()\n);\n\n-- Insert Default Roles\nINSERT INTO Roles (RoleName, Description) VALUES\n('Super Admin', 'Full system access'),\n('HR Admin', 'HR management access'),\n('Department Manager', 'Department level access'),\n('Employee', 'View only access');\n```\n\n### 3. UserRoles\n```sql\nCREATE TABLE UserRoles (\n    UserRoleId INT PRIMARY KEY IDENTITY(1,1),\n    UserId INT NOT NULL,\n    RoleId INT NOT NULL,\n    AssignedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (UserId) REFERENCES UserAccounts(UserId),\n    FOREIGN KEY (RoleId) REFERENCES Roles(RoleId),\n    UNIQUE(UserId, RoleId)\n);\n```\n\n### 4. Departments\n```sql\nCREATE TABLE Departments (\n    DepartmentId INT PRIMARY KEY IDENTITY(1,1),\n    DepartmentName NVARCHAR(255) NOT NULL UNIQUE,\n    DepartmentCode NVARCHAR(50) NOT NULL UNIQUE,\n    HeadId INT,\n    EstablishedDate DATE,\n    IsActive BIT NOT NULL DEFAULT 1,\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    UpdatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (HeadId) REFERENCES Employees(EmployeeId)\n);\n\nCREATE CLUSTERED INDEX IX_Departments_Code ON Departments(DepartmentCode);\n```\n\n### 5. Designations\n```sql\nCREATE TABLE Designations (\n    DesignationId INT PRIMARY KEY IDENTITY(1,1),\n    DesignationName NVARCHAR(255) NOT NULL UNIQUE,\n    DesignationCode NVARCHAR(50) NOT NULL UNIQUE,\n    DesignationLevel INT NOT NULL,\n    Description NVARCHAR(500),\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    UpdatedAt DATETIME NOT NULL DEFAULT GETUTCDATE()\n);\n\nCREATE CLUSTERED INDEX IX_Designations_Level ON Designations(DesignationLevel DESC);\n```\n\n### 6. Employees\n```sql\nCREATE TABLE Employees (\n    EmployeeId INT PRIMARY KEY IDENTITY(1,1),\n    EmployeeName NVARCHAR(255) NOT NULL,\n    EmployeeCode NVARCHAR(50) NOT NULL UNIQUE,\n    DepartmentId INT NOT NULL,\n    DesignationId INT NOT NULL,\n    Email NVARCHAR(255) NOT NULL UNIQUE,\n    Mobile NVARCHAR(20),\n    ReportToEmployeeId INT NULL,\n    JoiningDate DATE NOT NULL,\n    ProfilePicture NVARCHAR(MAX),\n    IsActive BIT NOT NULL DEFAULT 1,\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    UpdatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (DepartmentId) REFERENCES Departments(DepartmentId),\n    FOREIGN KEY (DesignationId) REFERENCES Designations(DesignationId),\n    FOREIGN KEY (ReportToEmployeeId) REFERENCES Employees(EmployeeId)\n);\n\nCREATE CLUSTERED INDEX IX_Employees_DepartmentId ON Employees(DepartmentId);\nCREATE NONCLUSTERED INDEX IX_Employees_ReportToEmployeeId ON Employees(ReportToEmployeeId);\nCREATE NONCLUSTERED INDEX IX_Employees_Code ON Employees(EmployeeCode);\n```\n\n### 7. EmployeeDesignationHistories\n```sql\nCREATE TABLE EmployeeDesignationHistories (\n    HistoryId INT PRIMARY KEY IDENTITY(1,1),\n    EmployeeId INT NOT NULL,\n    DesignationId INT NOT NULL,\n    FromDate DATE NOT NULL,\n    ToDate DATE,\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (EmployeeId) REFERENCES Employees(EmployeeId),\n    FOREIGN KEY (DesignationId) REFERENCES Designations(DesignationId)\n);\n```\n\n### 8. ManagerReportingRelationships\n```sql\nCREATE TABLE ManagerReportingRelationships (\n    RelationshipId INT PRIMARY KEY IDENTITY(1,1),\n    ManagerId INT NOT NULL,\n    ReportingEmployeeId INT NOT NULL,\n    EffectiveDate DATE NOT NULL,\n    EndDate DATE,\n    IsActive BIT NOT NULL DEFAULT 1,\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    UpdatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (ManagerId) REFERENCES Employees(EmployeeId),\n    FOREIGN KEY (ReportingEmployeeId) REFERENCES Employees(EmployeeId),\n    UNIQUE(ManagerId, ReportingEmployeeId, EffectiveDate)\n);\n```\n\n### 9. TransferHistories\n```sql\nCREATE TABLE TransferHistories (\n    TransferId INT PRIMARY KEY IDENTITY(1,1),\n    EmployeeId INT NOT NULL,\n    TransferType NVARCHAR(50) NOT NULL,\n    OldValue NVARCHAR(500),\n    NewValue NVARCHAR(500),\n    EffectiveDate DATE NOT NULL,\n    ChangedByUserId INT,\n    Reason NVARCHAR(MAX),\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (EmployeeId) REFERENCES Employees(EmployeeId),\n    FOREIGN KEY (ChangedByUserId) REFERENCES UserAccounts(UserId)\n);\n\nCREATE CLUSTERED INDEX IX_TransferHistories_EmployeeId ON TransferHistories(EmployeeId, EffectiveDate DESC);\n```\n\n### 10. Vacancies\n```sql\nCREATE TABLE Vacancies (\n    VacancyId INT PRIMARY KEY IDENTITY(1,1),\n    DepartmentId INT NOT NULL,\n    DesignationId INT NOT NULL,\n    PositionTitle NVARCHAR(255) NOT NULL,\n    ReportingToEmployeeId INT,\n    OpenedDate DATE NOT NULL,\n    ClosedDate DATE,\n    Status NVARCHAR(50) NOT NULL DEFAULT 'Open',\n    IsActive BIT NOT NULL DEFAULT 1,\n    CreatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    UpdatedAt DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    FOREIGN KEY (DepartmentId) REFERENCES Departments(DepartmentId),\n    FOREIGN KEY (DesignationId) REFERENCES Designations(DesignationId),\n    FOREIGN KEY (ReportingToEmployeeId) REFERENCES Employees(EmployeeId)\n);\n\nCREATE CLUSTERED INDEX IX_Vacancies_Status ON Vacancies(Status);\n```\n\n### 11. AuditLogs\n```sql\nCREATE TABLE AuditLogs (\n    AuditLogId BIGINT PRIMARY KEY IDENTITY(1,1),\n    UserId INT,\n    EntityType NVARCHAR(100) NOT NULL,\n    EntityId INT NOT NULL,\n    Action NVARCHAR(50) NOT NULL,\n    OldValues NVARCHAR(MAX),\n    NewValues NVARCHAR(MAX),\n    Timestamp DATETIME NOT NULL DEFAULT GETUTCDATE(),\n    IPAddress NVARCHAR(45),\n    FOREIGN KEY (UserId) REFERENCES UserAccounts(UserId)\n);\n\nCREATE CLUSTERED INDEX IX_AuditLogs_Timestamp ON AuditLogs(Timestamp DESC);\nCREATE NONCLUSTERED INDEX IX_AuditLogs_EntityType ON AuditLogs(EntityType, EntityId);\n```\n\n## Key Stored Procedures\n\n### Get Organization Hierarchy\n```sql\nCREATE PROCEDURE sp_GetOrganizationHierarchy\n    @TopEmployeeId INT = NULL\nAS\nBEGIN\n    WITH HierarchyCTE AS (\n        SELECT \n            EmployeeId,\n            EmployeeName,\n            EmployeeCode,\n            DepartmentId,\n            DesignationId,\n            ReportToEmployeeId,\n            1 AS HierarchyLevel\n        FROM Employees\n        WHERE (@TopEmployeeId IS NULL AND ReportToEmployeeId IS NULL)\n           OR (@TopEmployeeId IS NOT NULL AND EmployeeId = @TopEmployeeId)\n           AND IsActive = 1\n        \n        UNION ALL\n        \n        SELECT \n            e.EmployeeId,\n            e.EmployeeName,\n            e.EmployeeCode,\n            e.DepartmentId,\n            e.DesignationId,\n            e.ReportToEmployeeId,\n            h.HierarchyLevel + 1\n        FROM Employees e\n        INNER JOIN HierarchyCTE h ON e.ReportToEmployeeId = h.EmployeeId\n        WHERE e.IsActive = 1\n    )\n    SELECT \n        h.EmployeeId,\n        h.EmployeeName,\n        h.EmployeeCode,\n        d.DepartmentName,\n        des.DesignationName,\n        h.HierarchyLevel,\n        h.ReportToEmployeeId,\n        COUNT(DISTINCT e.EmployeeId) - 1 AS DirectReports\n    FROM HierarchyCTE h\n    LEFT JOIN Departments d ON h.DepartmentId = d.DepartmentId\n    LEFT JOIN Designations des ON h.DesignationId = des.DesignationId\n    LEFT JOIN Employees e ON h.EmployeeId = e.ReportToEmployeeId AND e.IsActive = 1\n    GROUP BY h.EmployeeId, h.EmployeeName, h.EmployeeCode, d.DepartmentName, \n             des.DesignationName, h.HierarchyLevel, h.ReportToEmployeeId\n    ORDER BY h.HierarchyLevel, h.EmployeeName;\nEND;\n```\n\n## Indexes Strategy\n\n**Primary Indexes:**\n- Clustered on EmployeeId in Employees table\n- Non-clustered on DepartmentId, ReportToEmployeeId\n- Non-clustered on Email for login queries\n\n**Analytics Indexes:**\n- DepartmentId + IsActive for department queries\n- DesignationLevel for hierarchy level queries\n- Timestamp for audit log retrieval\n\n**Query Optimization:**\n- Statistics auto-update enabled\n- Index hints where necessary\n- Materialized views for complex queries\n
